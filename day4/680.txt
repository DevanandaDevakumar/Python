class Solution:
    def validPalindrome(self, s: str) -> bool:
        i=0
        j=len(s)-1
        f=1
        while(i<j):
            if s[i].lower()!=s[j].lower():
                skip_left=s[i+1:j+1]
                skip_right=s[i:j]
                if skip_left[::-1]==skip_left or skip_right==skip_right[::-1]:
                    return True
                else:
                    return False
            i=i+1
            j=j-1
        return True
