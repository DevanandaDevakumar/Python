class Solution:
    def dailyTemperatures(self, temperatures: List[int]) -> List[int]:
        s=[]
        n=len(temperatures)
        res=[0]*n
        for i in range(n-1,-1,-1):
            while s and temperatures[i]>=temperatures[s[-1]]:
                s.pop()
            if s:
                res[i]=s[-1]-i
            s.append(i)
        return res
