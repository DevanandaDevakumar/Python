class Solution:
    def numSubarrayProductLessThanK(self, nums: List[int], k: int) -> int:
        if k<1:
            return 0
        l,r,p,c=0,0,1,0
        while r<len(nums):
            p=p*nums[r]
            while p>=k and l<=r:
                p=p//nums[l]
                l=l+1
            c=c+r-l+1
            r=r+1
        return c
