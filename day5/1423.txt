class Solution:
    def maxScore(self, cardPoints: List[int], k: int) -> int:
        s=sum(cardPoints[:k])
        m=s
        i=k-1
        j=len(cardPoints)-1
        while i>=0:
            s=s-cardPoints[i]+cardPoints[j]
            m=max(m,s)
            j=j-1
            i=i-1
        return m
